import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'
serve(async()=>{
  const supa=createClient(Deno.env.get('SUPABASE_URL'), Deno.env.get('SUPABASE_SERVICE_ROLE_KEY'))
  const tables=['customers','cards','market_listings','raw_orders','psa_submissions','member_reviews','site_settings']
  const backup={}
  for(const t of tables){ const {data}=await supa.from(t).select('*'); backup[t]=data }
  return new Response(JSON.stringify({ok:true, counts:Object.fromEntries(Object.entries(backup).map(([k,v])=>[k,v.length]))}), {headers:{'Content-Type':'application/json'}})
})
// Cron: 0 3 * * * ทุกวันตี 3
